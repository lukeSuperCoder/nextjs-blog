```typescript
/**
 * 工具函数库
 *
 * 这个文件包含了项目中常用的工具函数
 * 主要用于类名合并、格式化等通用操作
 */

import { type ClassValue, clsx } from 'clsx'
import { twMerge } from 'tailwind-merge'

/**
 * cn - 类名合并工具函数
 *
 * @description
 * 这个函数结合了 clsx 和 tailwind-merge 的功能
 *
 * @技术原理:
 * 1. clsx: 用于条件性地组合类名,支持多种输入格式
 * 2. tailwind-merge: 智能合并 Tailwind CSS 类名,避免冲突
 *
 * @example
 * cn('px-2 py-1', condition && 'bg-blue-500')
 * cn('px-2', 'px-4') // 输出 'px-4',后者覆盖前者
 *
 * @最佳实践:
 * - 在 Tailwind 项目中,这是标准的类名合并方式
 * - 避免了类名冲突问题(如 px-2 和 px-4 同时存在)
 */
export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs))
}

/**
 * 格式化日期
 *
 * @param date - 日期字符串或 Date 对象
 * @returns 格式化后的日期字符串
 */
export function formatDate(date: string | Date): string {
  const d = new Date(date)
  return d.toLocaleDateString('zh-CN', {
    year: 'numeric',
    month: 'long',
    day: 'numeric',
  })
}

/**
 * 截断文本
 *
 * @param text - 要截断的文本
 * @param length - 最大长度
 * @returns 截断后的文本
 */
export function truncate(text: string, length: number): string {
  if (text.length <= length) return text
  return text.slice(0, length) + '...'
}
```

#### 2.2 定义数据模型

创建 `prisma/schema.prisma`:

```prisma
// Prisma Schema 文件
// 这个文件定义了数据库的结构和关系

// 数据源配置
// provider: 指定数据库类型 (postgresql)
// url: 从环境变量读取数据库连接字符串
datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// 客户端生成器配置
// 这会生成 Prisma Client,用于在代码中操作数据库
generator client {
  provider = "prisma-client-js"
}

// ============================================
// 用户模型
// ============================================

/**
 * User 模型 - 用户表
 *
 * @技术原理:
 * 1. @id: 主键标识
 * 2. @default(cuid()): 使用 cuid 生成唯一 ID
 * 3. @unique: 唯一约束,确保邮箱不重复
 * 4. @default(now()): 自动设置创建时间
 * 5. @updatedAt: 自动更新修改时间
 *
 * @关系:
 * - 一个用户可以有多篇文章 (一对多)
 * - 一个用户可以有多个 Account (OAuth 账户)
 * - 一个用户可以有多个 Session (登录会话)
 */
model User {
  id            String    @id @default(cuid())
  name          String?
  email         String    @unique
  emailVerified DateTime?
  image         String?
  password      String?   // 密码哈希值,使用 bcrypt 加密
  role          String    @default("visitor") // admin, visitor

  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  // 关系字段
  posts         Post[]     // 用户创建的文章
  accounts      Account[]  // OAuth 账户
  sessions      Session[]  // 登录会话
}

// ============================================
// NextAuth.js 相关模型
// ============================================

/**
 * Account 模型 - OAuth 账户表
 *
 * @说明:
 * NextAuth.js 使用这个表存储 OAuth 提供商的账户信息
 * 例如: GitHub、Google 等第三方登录
 */
model Account {
  id                String  @id @default(cuid())
  userId            String
  type              String
  provider          String
  providerAccountId String
  refresh_token     String?
  access_token      String?
  expires_at        Int?
  token_type        String?
  scope             String?
  id_token          String?
  session_state     String?

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([provider, providerAccountId])
}

/**
 * Session 模型 - 会话表
 *
 * @说明:
 * 存储用户的登录会话信息
 */
model Session {
  id           String   @id @default(cuid())
  sessionToken String   @unique
  userId       String
  expires      DateTime

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)
}

/**
 * VerificationToken 模型 - 验证令牌表
 *
 * @说明:
 * 用于邮箱验证、密码重置等场景
 */
model VerificationToken {
  identifier String
  token      String   @unique
  expires    DateTime

  @@unique([identifier, token])
}

// ============================================
// 文章模型
// ============================================

/**
 * Post 模型 - 文章表
 *
 * @技术原理:
 * 1. slug: URL 友好的标识符,如 "hello-world"
 * 2. @db.Text: SQLite 中用于存储大文本
 * 3. status: 文章状态 (draft: 草稿, published: 已发布)
 * 4. views, likes: 文章统计数据
 *
 * @关系:
 * - 多对一: 文章属于一个用户
 * - 多对一: 文章属于一个分类
 * - 多对多: 文章可以有多个标签
 */
model Post {
  id          String    @id @default(cuid())
  title       String
  slug        String    @unique
  content     String    @db.Text  // PostgreSQL: TEXT 类型,无长度限制
  excerpt     String?   @db.VarChar(500)  // 摘要限制 500 字符
  coverImage  String?   @db.VarChar(500)  // URL 限制 500 字符
  status      String    @default("draft") // draft, published
  views       Int       @default(0)
  likes       Int       @default(0)

  // SEO 相关字段
  metaTitle       String?  @db.VarChar(100)
  metaDescription String?  @db.VarChar(200)
  metaKeywords    String?  @db.VarChar(200)

  publishedAt DateTime?
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt

  // 外键关系
  authorId    String
  author      User      @relation(fields: [authorId], references: [id], onDelete: Cascade)

  categoryId  String?
  category    Category? @relation(fields: [categoryId], references: [id], onDelete: SetNull)

  tags        Tag[]     // 多对多关系

  @@index([status])      // 为 status 创建索引,加快查询
  @@index([publishedAt]) // 为发布时间创建索引
  @@index([authorId])    // 为作者 ID 创建索引
}

/**
 * PostgreSQL 字段类型说明:
 *
 * @db.Text: 无长度限制的文本,适合存储大量内容
 * @db.VarChar(n): 可变长度字符串,最大 n 个字符
 * @db.Char(n): 固定长度字符串
 *
 * 为什么使用类型注解:
 * 1. 优化数据库性能
 * 2. 节省存储空间
 * 3. 提供更好的数据验证
 */

// ============================================
// 分类模型
// ============================================

/**
 * Category 模型 - 分类表
 *
 * @技术原理:
 * 1. 一个文章只能属于一个分类
 * 2. slug 用于生成友好的 URL
 * 3. description 用于 SEO 和展示
 */
model Category {
  id          String   @id @default(cuid())
  name        String   @unique
  slug        String   @unique
  description String?
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  posts       Post[]
}

// ============================================
// 标签模型
// ============================================

/**
 * Tag 模型 - 标签表
 *
 * @技术原理:
 * 1. 一个文章可以有多个标签 (多对多关系)
 * 2. color 用于在 UI 中显示不同颜色的标签
 */
model Tag {
  id        String   @id @default(cuid())
  name      String   @unique
  slug      String   @unique
  color     String?  // 十六进制颜色值,如 "#3B82F6"
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  posts     Post[]
}
```

- Prisma Schema 语法
- 数据模型设计原则
- 一对多、多对多关系

#### 2.3 创建 Prisma Client

创建 `src/lib/prisma.ts`:

```typescript
/**
 * Prisma Client 单例
 *
 * @技术原理:
 * 在开发环境中,Next.js 的热重载会导致多次创建 Prisma Client 实例
 * 这会导致数据库连接数过多的问题
 *
 * @解决方案:
 * 使用单例模式,确保只创建一个 Prisma Client 实例
 * 在开发环境中,将实例存储在 global 对象上,避免热重载时重复创建
 *
 * @最佳实践:
 * 这是 Prisma 官方推荐的做法
 */

import { PrismaClient } from '@prisma/client'

// 声明全局类型
declare global {
  // eslint-disable-next-line no-var
  var prisma: PrismaClient | undefined
}

/**
 * 创建 Prisma Client 实例
 *
 * @说明:
 * - 在生产环境: 直接创建新实例
 * - 在开发环境: 使用 global 缓存,避免热重载时重复创建
 */
export const prisma = global.prisma || new PrismaClient()

// 在开发环境中,将实例存储到 global 对象
// 这样热重载时可以复用同一个实例
if (process.env.NODE_ENV !== 'production') {
  global.prisma = prisma
}

/**
 * 为什么需要这样做?
 *
 * 1. Next.js 开发环境的热重载:
 *    每次代码更改,模块会重新加载,导致新的 Prisma Client 被创建
 *
 * 2. 数据库连接限制:
 *    每个 Prisma Client 实例会创建一个连接池
 *    过多的实例会导致数据库连接数超限
 *
 * 3. global 对象的作用:
 *    global 对象在 Node.js 中是全局的,不受模块重载影响
 *    存储在 global 上的对象在热重载时会被保留
 *
 * 4. 生产环境:
 *    生产环境不会有热重载,所以直接创建实例即可
 */
```

```typescript
/**
 * NextAuth.js 配置文件
 *
 * @技术原理:
 * NextAuth.js 是 Next.js 的认证解决方案,支持:
 * 1. 多种认证提供商 (OAuth, Email, Credentials)
 * 2. Session 管理
 * 3. JWT 或数据库 Session
 * 4. CSRF 保护
 *
 * @核心概念:
 * - Providers: 认证提供商,如 GitHub、Google、邮箱密码等
 * - Adapter: 数据库适配器,用于存储用户、会话等数据
 * - Callbacks: 生命周期钩子,用于自定义认证流程
 * - Session Strategy: JWT 或 Database
 */

import { NextAuthOptions } from 'next-auth'
import { PrismaAdapter } from '@auth/prisma-adapter'
import CredentialsProvider from 'next-auth/providers/credentials'
import bcrypt from 'bcryptjs'
import { prisma } from './prisma'

/**
 * NextAuth 配置对象
 *
 * @说明:
 * 这个配置定义了认证系统的所有行为
 */
export const authOptions: NextAuthOptions = {
  /**
   * Adapter: 数据库适配器
   *
   * @作用:
   * 将 NextAuth 的数据存储到 Prisma 数据库
   * 自动处理用户、账户、会话等表的操作
   */
  adapter: PrismaAdapter(prisma),

  /**
   * Session Strategy: 会话策略
   *
   * @选项:
   * - "jwt": 使用 JWT 存储会话信息 (无状态,更快)
   * - "database": 使用数据库存储会话信息 (有状态,更安全)
   *
   * @当前选择: JWT
   * @原因:
   * 1. 性能更好,不需要查询数据库
   * 2. 更适合无服务器环境 (Vercel)
   * 3. 水平扩展更容易
   */
  session: {
    strategy: 'jwt',
  },

  /**
   * Pages: 自定义认证页面
   *
   * @说明:
   * 指定登录、错误等页面的路径
   * 如不指定,NextAuth 会使用默认页面
   */
  pages: {
    signIn: '/auth/signin',
    error: '/auth/error',
  },

  /**
   * Providers: 认证提供商
   *
   * @说明:
   * 定义支持的登录方式
   */
  providers: [
    /**
     * Credentials Provider: 邮箱密码登录
     *
     * @技术原理:
     * 1. 用户提交邮箱和密码
     * 2. 在 authorize 函数中验证
     * 3. 返回用户对象或 null
     *
     * @安全性:
     * - 密码使用 bcrypt 加密存储
     * - bcrypt 是一个慢哈希函数,防止暴力破解
     * - 包含 salt,每个密码的哈希值都不同
     */
    CredentialsProvider({
      name: 'credentials',
      credentials: {
        email: { label: 'Email', type: 'email' },
        password: { label: 'Password', type: 'password' },
      },

      /**
       * authorize 函数: 验证用户凭证
       *
       * @参数:
       * - credentials: 用户提交的凭证
       *
       * @返回值:
       * - 成功: 返回用户对象
       * - 失败: 返回 null
       *
       * @流程:
       * 1. 根据邮箱查询用户
       * 2. 验证密码
       * 3. 返回用户信息
       */
      async authorize(credentials) {
        // 验证输入
        if (!credentials?.email || !credentials?.password) {
          throw new Error('请输入邮箱和密码')
        }

        // 查询用户
        const user = await prisma.user.findUnique({
          where: { email: credentials.email },
        })

        if (!user || !user.password) {
          throw new Error('用户不存在')
        }

        // 验证密码
        // bcrypt.compare 会自动处理 salt
        const isPasswordValid = await bcrypt.compare(
          credentials.password,
          user.password
        )

        if (!isPasswordValid) {
          throw new Error('密码错误')
        }

        // 返回用户对象
        // NextAuth 会将这个对象存入 JWT 或数据库
        return {
          id: user.id,
          email: user.email,
          name: user.name,
          role: user.role,
          image: user.image,
        }
      },
    }),

    /**
     * 可以添加其他提供商,如:
     *
     * GitHubProvider({
     *   clientId: process.env.GITHUB_ID,
     *   clientSecret: process.env.GITHUB_SECRET,
     * }),
     */
  ],

  /**
   * Callbacks: 生命周期钩子
   *
   * @说明:
   * 在认证流程的关键节点执行自定义逻辑
   */
  callbacks: {
    /**
     * jwt callback: JWT 生成时调用
     *
     * @触发时机:
     * 1. 用户登录时
     * 2. 使用 getSession 或 getToken 时
     *
     * @作用:
     * 将额外信息添加到 JWT 中
     *
     * @参数:
     * - token: JWT 对象
     * - user: 用户对象 (仅登录时存在)
     *
     * @注意:
     * JWT 有大小限制 (4KB),不要存储过多数据
     */
    async jwt({ token, user }) {
      // 首次登录时,user 对象存在
      // 将用户信息添加到 token
      if (user) {
        token.id = user.id
        token.role = user.role
      }
      return token
    },

    /**
     * session callback: Session 对象生成时调用
     *
     * @触发时机:
     * 使用 getServerSession 或 useSession 时
     *
     * @作用:
     * 将 JWT 中的信息添加到 Session 对象
     * 这个 Session 对象会传递给客户端
     *
     * @参数:
     * - session: Session 对象
     * - token: JWT 对象
     *
     * @安全性:
     * 不要将敏感信息添加到 session,它会发送到客户端
     */
    async session({ session, token }) {
      if (session.user) {
        session.user.id = token.id as string
        session.user.role = token.role as string
      }
      return session
    },
  },

  /**
   * Secret: JWT 签名密钥
   *
   * @重要性:
   * 这个密钥用于签名 JWT,必须保密
   * 泄露会导致 JWT 被伪造
   *
   * @生成方式:
   * openssl rand -base64 32
   */
  secret: process.env.NEXTAUTH_SECRET,
}
```

```typescript
        </div>

/**

 *
```

```typescript
 * 3. 可以有多个路由组
 *
 * @示例:
 * (frontend)/page.tsx -> /
 * (frontend)/posts/page.tsx -> /posts
 * (dashboard)/page.tsx -> /dashboard (不同的布局)

import { Header } from '@/components/layout/header'
import { Footer } from '@/components/layout/footer'

interface LayoutProps {
  children: React.ReactNode
}

/**
 * 前台布局组件
 *
 * @说明:
 * 所有前台页面都会使用这个布局
 * 包含 Header 和 Footer
 */
export default function FrontendLayout({ children }: LayoutProps) {
  return (
    /**
     * 布局结构
     *
     * @技术:
     * 使用 Flexbox 实现 Sticky Footer
     * min-h-screen: 最小高度为视口高度
```

#### 4.4 创建首页

创建 `src/app/(frontend)/page.tsx`:

```typescript
/**
 * 首页组件
 *
 * @渲染策略: SSG (Static Site Generation)
 *
 * @技术原理:
 * 1. 这是一个服务端组件,可以直接访问数据库
 * 2. 在构建时生成静态 HTML
 * 3. 无需 API 路由,直接在组件中获取数据
 *
 * @性能优势:
 * 1. 构建时生成,运行时无需查询数据库
 * 2. 可以部署到 CDN
 * 3. 首屏加载速度快
 *
 * @SEO 优势:
 * 1. 静态 HTML,搜索引擎可以直接抓取
 * 2. 无需等待 JavaScript 执行
 */

import Link from 'next/link'
import { prisma } from '@/lib/prisma'
import { Button } from '@/components/ui/button'
import { formatDate } from '@/lib/utils'

/**
 * 获取最新文章
 *
 * @说明:
 * 这是一个异步函数,在服务端执行
 *
 * @技术原理:
 * 1. 直接使用 Prisma 查询数据库
 * 2. 不需要 API 路由
 * 3. 查询结果在构建时就确定了 (SSG)
 *
 * @Prisma 查询:
 * - where: 查询条件 (只要已发布的文章)
 * - orderBy: 排序 (按发布时间倒序)
 * - take: 限制数量
 * - include: 关联查询 (包含作者、分类、标签)
 */
async function getLatestPosts() {
  const posts = await prisma.post.findMany({
    where: {
      status: 'published',
    },
    orderBy: {
      publishedAt: 'desc',
    },
    take: 6,
    include: {
      author: {
        select: {
          name: true,
          image: true,
        },
      },
      category: true,
      tags: true,
    },
  })

  return posts
}

/**
 * 首页组件
 *
 * @说明:
 * 这是一个异步的服务端组件
 *
 * @async/await:
 * Next.js 13+ 支持异步的服务端组件
 * 可以在组件内部直接使用 await
 *
 * @注意:
 * 这个特性只在服务端组件中可用
 * 客户端组件不能是异步的
 */
export default async function HomePage() {
  /**
   * 获取数据
   *
   * @时机:
   * - SSG: 在构建时执行
   * - SSR: 在每次请求时执行
   * - ISR: 在首次请求或重新验证时执行
   *
   * @当前策略: SSG
   * 因为没有设置 revalidate 或使用 fetch cache 选项
   */
  const posts = await getLatestPosts()

  return (
    <div className="container py-12">
      {/* Hero 区域 */}
      <section className="flex flex-col items-center text-center space-y-6 py-12 md:py-24">
        <h1 className="text-4xl font-bold tracking-tighter sm:text-5xl md:text-6xl lg:text-7xl">
          欢迎来到我的博客
        </h1>
        <p className="max-w-[700px] text-lg text-muted-foreground sm:text-xl">
          分享 Web 开发技术,记录学习和成长
        </p>
        <div className="flex gap-4">
          <Link href="/posts">
            <Button size="lg">浏览文章</Button>
          </Link>
          <Link href="/about">
            <Button size="lg" variant="outline">
              关于我
            </Button>
          </Link>
        </div>
      </section>

      {/* 最新文章 */}
      <section className="py-12">
        <div className="flex items-center justify-between mb-8">
          <h2 className="text-3xl font-bold">最新文章</h2>
          <Link href="/posts">
            <Button variant="ghost">查看全部 →</Button>
          </Link>
        </div>

        {/**
          * 文章网格布局
          *
          * @响应式:
          * - grid-cols-1: 移动端单列
          * - md:grid-cols-2: 平板双列
          * - lg:grid-cols-3: 桌面三列
          */}
        <div className="grid gap-6 md:grid-cols-2 lg:grid-cols-3">
          {posts.map((post) => (
            <article
              key={post.id}
              className="group flex flex-col space-y-4 border rounded-lg p-6 hover:shadow-lg transition-shadow"
            >
              {/* 分类标签 */}
              {post.category && (
                <div className="text-sm text-primary font-medium">
                  {post.category.name}
                </div>
              )}

              {/* 标题 */}
              <Link href={`/posts/${post.slug}`}>
                <h3 className="text-xl font-bold line-clamp-2 group-hover:text-primary transition-colors">
                  {post.title}
                </h3>
              </Link>

              {/* 摘要 */}
              {post.excerpt && (
                <p className="text-muted-foreground line-clamp-3">
                  {post.excerpt}
                </p>
              )}

              {/* 标签 */}
              {post.tags.length > 0 && (
                <div className="flex flex-wrap gap-2">
                  {post.tags.map((tag) => (
                    <span
                      key={tag.id}
                      className="text-xs px-2 py-1 bg-secondary rounded-md"
                    >
                      {tag.name}
                    </span>
                  ))}
                </div>
              )}

              {/* 元信息 */}
              <div className="flex items-center text-sm text-muted-foreground">
                <time dateTime={post.publishedAt?.toISOString()}>
                  {post.publishedAt && formatDate(post.publishedAt)}
                </time>
                <span className="mx-2">·</span>
                <span>{post.views} 次阅读</span>
              </div>
            </article>
          ))}
        </div>
      </section>
    </div>
  )
}

/**
 * 关键学习点:
 *
 * 1. 服务端组件的优势:
 *    - 直接访问数据库,无需 API
 *    - 减少客户端 JavaScript
 *    - 更好的 SEO
 *    - 更快的首屏加载
 *
 * 2. SSG (Static Site Generation):
 *    - 在构建时生成 HTML
 *    - 适合内容不经常变化的页面
 *    - 可以配合 ISR 使用
 *
 * 3. Prisma 关联查询:
 *    - include: 包含关联数据
 *    - select: 选择特定字段
 *    - 自动处理 JOIN 操作
 *
 * 4. Tailwind CSS 实用类:
 *    - line-clamp-*: 限制行数
 *    - group-hover: 组悬停效果
 *    - transition-*: 过渡动画
 *
 * 5. 语义化 HTML:
 *    - <article>: 独立的内容单元
 *    - <time>: 时间信息
 *    - dateTime: 机器可读的时间格式
 */
```
