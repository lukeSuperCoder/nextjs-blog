```typescript
/**
 * 文章列表页
 *
 * @渲染策略: ISR (Incremental Static Regeneration)
 *
 * @技术原理:
 * 1. 首次访问时生成静态页面
 * 2. 定时重新生成 (revalidate)
 * 3. 保持静态页面的性能,又能更新内容
 *
 * @优势:
 * - 快速的页面加载 (静态)
 * - 内容保持更新 (定时重新生成)
 * - 无需每次请求都查询数据库
 *
 * @ISR vs SSR vs SSG:
 * - SSG: 构建时生成,内容固定
 * - SSR: 每次请求都生成,最新但慢
 * - ISR: 结合两者优势,定时更新
 */

import Link from 'next/link'
import { prisma } from '@/lib/prisma'
import { Button } from '@/components/ui/button'
import { Badge } from '@/components/ui/badge'
import { formatDate } from '@/lib/utils'

/**
 * 页面组件的 Props 类型
 *
 * @说明:
 * Next.js 会将 URL 参数传递给页面组件
 *
 * @参数:
 * - searchParams: URL 查询参数 (如 ?page=2&category=react)
 */
interface PageProps {
  searchParams: {
    page?: string
    category?: string
    tag?: string
  }
}

/**
 * ISR 配置
 *
 * @说明:
 * 导出 revalidate 变量设置重新验证时间
 *
 * @技术原理:
 * Next.js 会读取这个导出值
 * 配置页面的缓存策略
 *
 * @时间单位: 秒
 * @示例: 60 = 1分钟后重新生成页面
 */
export const revalidate = 60 // 60秒后重新验证

/**
 * 动态 Metadata
 *
 * @说明:
 * 根据页面内容生成不同的 meta 标签
 *
 * @技术原理:
 * 1. generateMetadata 是异步函数
 * 2. 可以访问 props (包括 searchParams)
 * 3. 返回 Metadata 对象
 */
export async function generateMetadata({ searchParams }: PageProps) {
  const page = Number(searchParams.page) || 1

  return {
    title: `文章列表 - 第 ${page} 页`,
    description: '浏览所有技术文章',
  }
}

/**
 * 获取文章列表
 *
 * @参数:
 * - page: 当前页码
 * - pageSize: 每页文章数
 * - category: 分类筛选
 * - tag: 标签筛选
 */
async function getPosts(
  page: number = 1,
  pageSize: number = 10,
  category?: string,
  tag?: string
) {
  /**
   * 构建查询条件
   *
   * @技术原理:
   * Prisma 的条件查询
   * 使用对象组合构建复杂查询
   */
  const where: any = {
    status: 'published',
  }

  // 分类筛选
  if (category) {
    where.category = {
      slug: category,
    }
  }

  // 标签筛选
  if (tag) {
    where.tags = {
      some: {
        slug: tag,
      },
    }
  }

  /**
   * 并行查询
   *
   * @技术原理:
   * Promise.all 同时执行多个异步操作
   * 比串行执行更快
   *
   * @查询:
   * 1. 查询文章列表
   * 2. 查询文章总数 (用于分页)
   */
  const [posts, total] = await Promise.all([
    /**
     * findMany: 查询多条记录
     *
     * @参数:
     * - where: 查询条件
     * - skip: 跳过的记录数 (分页)
     * - take: 获取的记录数 (每页数量)
     * - orderBy: 排序方式
     * - include: 关联查询
     */
    prisma.post.findMany({
      where,
      skip: (page - 1) * pageSize,
      take: pageSize,
      orderBy: {
        publishedAt: 'desc',
      },
      include: {
        author: {
          select: {
            name: true,
            image: true,
          },
        },
        category: true,
        tags: true,
        // 使用 select 避免查询大字段
        // content 字段很大,列表页不需要
      },
      // 注意:我们没有 select content,节省带宽
    }),

    /**
     * count: 统计记录数
     *
     * @说明:
     * 用于计算总页数
     */
    prisma.post.count({ where }),
  ])

  /**
   * 计算分页信息
   */
  const totalPages = Math.ceil(total / pageSize)
  const hasNextPage = page < totalPages
  const hasPrevPage = page > 1

  return {
    posts,
    pagination: {
      total,
      page,
      pageSize,
      totalPages,
      hasNextPage,
      hasPrevPage,
    },
  }
}

/**
 * 文章列表页组件
 */
export default async function PostsPage({ searchParams }: PageProps) {
  const page = Number(searchParams.page) || 1
  const category = searchParams.category
  const tag = searchParams.tag

  /**
   * 获取数据
   *
   * @说明:
   * 这是一个异步的服务端组件
   * 可以直接使用 await
   */
  const { posts, pagination } = await getPosts(page, 10, category, tag)

  return (
    <div className="container py-12">
      <div className="max-w-4xl mx-auto">
        {/* 页面标题 */}
        <div className="mb-8">
          <h1 className="text-4xl font-bold mb-4">
            {category ? `分类: ${category}` : tag ? `标签: ${tag}` : '所有文章'}
          </h1>
          <p className="text-muted-foreground">
            共 {pagination.total} 篇文章
          </p>
        </div>

        {/* 文章列表 */}
        {posts.length > 0 ? (
          <div className="space-y-8">
            {posts.map((post) => (
              /**
               * 文章卡片
               *
               * @key:
               * React 列表渲染必须提供 key
               * 帮助 React 识别哪些项发生了变化
               *
               * @最佳实践:
               * 使用唯一且稳定的 ID,不要使用索引
               */
              <article
                key={post.id}
                className="border-b pb-8 last:border-b-0"
              >
                {/* 分类标签 */}
                {post.category && (
                  <Link href={`/posts?category=${post.category.slug}`}>
                    <Badge variant="secondary" className="mb-2">
                      {post.category.name}
                    </Badge>
                  </Link>
                )}

                {/* 文章标题 */}
                <Link href={`/posts/${post.slug}`}>
                  <h2 className="text-2xl font-bold mb-2 hover:text-primary transition-colors">
                    {post.title}
                  </h2>
                </Link>

                {/* 文章摘要 */}
                {post.excerpt && (
                  <p className="text-muted-foreground mb-4 line-clamp-2">
                    {post.excerpt}
                  </p>
                )}

                {/* 标签 */}
                {post.tags.length > 0 && (
                  <div className="flex flex-wrap gap-2 mb-4">
                    {post.tags.map((tag) => (
                      <Link
                        key={tag.id}
                        href={`/posts?tag=${tag.slug}`}
                      >
                        <Badge variant="outline" className="hover:bg-secondary">
                          {tag.name}
                        </Badge>
                      </Link>
                    ))}
                  </div>
                )}

                {/* 元信息 */}
                <div className="flex items-center gap-4 text-sm text-muted-foreground">
                  <time dateTime={post.publishedAt?.toISOString()}>
                    {post.publishedAt && formatDate(post.publishedAt)}
                  </time>
                  <span>·</span>
                  <span>{post.views} 次阅读</span>
                  <span>·</span>
                  <span>{post.likes} 个赞</span>
                </div>
              </article>
            ))}
          </div>
        ) : (
          <div className="text-center py-12">
            <p className="text-muted-foreground">暂无文章</p>
          </div>
        )}

        {/* 分页 */}
        {pagination.totalPages > 1 && (
          /**
            * 分页组件
            *
            * @技术原理:
            * 使用 URL 参数实现分页
            * 这样分页状态可以被分享和书签
            *
            * @最佳实践:
            * - 分页信息在 URL 中
            * - 支持浏览器前进后退
            * - SEO 友好
            */
          <div className="flex items-center justify-center gap-2 mt-12">
            {/* 上一页 */}
            {pagination.hasPrevPage && (
              <Link href={`/posts?page=${page - 1}`}>
                <Button variant="outline">上一页</Button>
              </Link>
            )}

            {/* 页码 */}
            <div className="flex items-center gap-2">
              {Array.from({ length: pagination.totalPages }, (_, i) => i + 1)
                .filter((pageNum) => {
                  // 显示当前页附近的页码
                  return (
                    pageNum === 1 ||
                    pageNum === pagination.totalPages ||
                    Math.abs(pageNum - page) <= 2
                  )
                })
                .map((pageNum, index, array) => {
                  // 添加省略号
                  const prevPageNum = array[index - 1]
                  const showEllipsis = prevPageNum && pageNum - prevPageNum > 1

                  return (
                    <>
                      {showEllipsis && <span key={`ellipsis-${pageNum}`}>...</span>}
                      <Link key={pageNum} href={`/posts?page=${pageNum}`}>
                        <Button
                          variant={pageNum === page ? 'default' : 'outline'}
                        >
                          {pageNum}
                        </Button>
                      </Link>
                    </>
                  )
                })}
            </div>

            {/* 下一页 */}
            {pagination.hasNextPage && (
              <Link href={`/posts?page=${page + 1}`}>
                <Button variant="outline">下一页</Button>
              </Link>
            )}
          </div>
        )}
      </div>
    </div>
  )
}

/**
 * 关键学习点:
 *
 * 1. ISR (Incremental Static Regeneration):
 *    - 静态生成 + 定时更新
 *    - 最佳性能和内容新鲜度的平衡
 *    - 使用 revalidate 配置
 *
 * 2. Prisma 查询优化:
 *    - 使用 select 避免查询不需要的字段
```

#### 5.2 创建文章详情页

```typescript
/**
 * 文章详情页
 *
 * @渲染策略: SSG + ISR
 *
 * @技术原理:
 * 1. 使用 generateStaticParams 预渲染页面
 * 2. 构建时生成所有文章的静态页面
 * 3. 新文章通过 fallback 和 revalidate 生成
 *
 * @动态路由:
 * [slug] 是动态参数
 * /posts/hello-world -> params.slug === 'hello-world'
 */

import { notFound } from 'next/navigation'
import { prisma } from '@/lib/prisma'
import { formatDate } from '@/lib/utils'
import { MarkdownRenderer } from '@/components/post/markdown-renderer'
import { Badge } from '@/components/ui/badge'
import Link from 'next/link'

/**
 * 页面 Props 类型
 */
interface PageProps {
  params: {
    slug: string
  }
}

/**
 * ISR 配置
 */
export const revalidate = 3600 // 1小时

/**
 * generateStaticParams: 生成静态路径
 *
 * @技术原理:
 * 1. 在构建时调用
 * 2. 返回要预渲染的参数列表
 * 3. Next.js 会为每个参数生成静态页面
 *
 * @优势:
 * - 构建时生成,无需运行时查询
 * - 极快的页面加载速度
 * - SEO 友好
 *
 * @返回格式:
 * [{ slug: 'post-1' }, { slug: 'post-2' }]
 */
export async function generateStaticParams() {
  /**
   * 获取所有已发布文章的 slug
   *
   * @注意:
   * 只查询 slug 字段,不查询其他内容
   * 减少构建时的数据传输
   */
  const posts = await prisma.post.findMany({
    where: {
      status: 'published',
    },
    select: {
      slug: true,
    },
  })

  /**
   * 返回参数数组
   *
   * @格式:
   * [{ slug: 'hello' }, { slug: 'world' }]
   */
  return posts.map((post) => ({
    slug: post.slug,
  }))
}

/**
 * 动态 Metadata
 *
 * @说明:
 * 为每篇文章生成独特的 SEO 信息
 */
export async function generateMetadata({ params }: PageProps) {
  const post = await prisma.post.findUnique({
    where: { slug: params.slug },
    select: {
      title: true,
      excerpt: true,
      coverImage: true,
      metaTitle: true,
      metaDescription: true,
      metaKeywords: true,
    },
  })

  if (!post) {
    return {
      title: '文章不存在',
    }
  }

  return {
    title: post.metaTitle || post.title,
    description: post.metaDescription || post.excerpt,
    keywords: post.metaKeywords?.split(','),
    /**
     * Open Graph: 社交媒体分享
     *
     * @说明:
     * 当文章被分享到 Facebook、Twitter 等平台时
     * 会显示这些信息
     */
    openGraph: {
      title: post.title,
      description: post.excerpt || undefined,
      images: post.coverImage ? [post.coverImage] : undefined,
      type: 'article',
    },
    /**
     * Twitter Card
     *
     * @说明:
     * Twitter 特定的分享卡片
     */
    twitter: {
      card: 'summary_large_image',
      title: post.title,
      description: post.excerpt || undefined,
      images: post.coverImage ? [post.coverImage] : undefined,
    },
  }
}

/**
 * 获取文章详情
 */
async function getPost(slug: string) {
  /**
   * findUnique: 查询单条记录
   *
   * @说明:
   * 根据唯一字段查询
   * slug 在数据库中是唯一的
   */
  const post = await prisma.post.findUnique({
    where: { slug },
    include: {
      author: {
        select: {
          name: true,
          image: true,
        },
      },
      category: true,
      tags: true,
    },
  })

  return post
}

/**
 * 增加文章浏览量
 *
 * @说明:
 * 这是一个副作用操作
 * 每次访问页面时增加浏览量
 */
async function incrementViews(slug: string) {
  await prisma.post.update({
    where: { slug },
    data: {
      /**
       * increment: 自增操作
       *
       * @技术原理:
       * Prisma 的原子操作
       * 直接在数据库层面执行 +1
       * 避免并发问题
       *
       * @SQL 等效:
       * UPDATE posts SET views = views + 1 WHERE slug = ?
       */
      views: {
        increment: 1,
      },
    },
  })
}

/**
 * 文章详情页组件
 */
export default async function PostPage({ params }: PageProps) {
  /**
   * 获取文章数据
   */
  const post = await getPost(params.slug)

  /**
   * 处理文章不存在
   *
   * @技术原理:
   * notFound() 会触发 404 页面
   * Next.js 会渲染 not-found.tsx
   */
  if (!post || post.status !== 'published') {
    notFound()
  }

  /**
   * 增加浏览量
   *
   * @注意:
   * 这是一个副作用
   * 在生产环境中,考虑使用:
   * 1. 客户端 API 调用 (更准确)
   * 2. 中间件统计 (更高效)
   * 3. 分析服务 (Google Analytics)
   */
  await incrementViews(params.slug)

  return (
    <article className="container py-12">
      <div className="max-w-3xl mx-auto">
        {/* 文章头部 */}
        <header className="mb-8">
          {/* 分类 */}
          {post.category && (
            <Link href={`/posts?category=${post.category.slug}`}>
              <Badge className="mb-4">{post.category.name}</Badge>
            </Link>
          )}

          {/* 标题 */}
          <h1 className="text-4xl font-bold mb-4">{post.title}</h1>

          {/* 元信息 */}
          <div className="flex items-center gap-4 text-sm text-muted-foreground">
            <time dateTime={post.publishedAt?.toISOString()}>
              {post.publishedAt && formatDate(post.publishedAt)}
            </time>
            <span>·</span>
            <span>{post.views} 次阅读</span>
            <span>·</span>
            <span>{post.likes} 个赞</span>
          </div>

          {/* 标签 */}
          {post.tags.length > 0 && (
            <div className="flex flex-wrap gap-2 mt-4">
              {post.tags.map((tag) => (
                <Link key={tag.id} href={`/posts?tag=${tag.slug}`}>
                  <Badge variant="outline">{tag.name}</Badge>
                </Link>
              ))}
            </div>
          )}
        </header>

        {/* 封面图 */}
        {post.coverImage && (
          <div className="mb-8">
            <img
              src={post.coverImage}
              alt={post.title}
              className="w-full rounded-lg"
            />
          </div>
        )}

        {/* 文章内容 */}
        <div className="prose prose-lg dark:prose-invert max-w-none">
          {/**
            * MarkdownRenderer 组件
            *
            * @说明:
            * 渲染 Markdown 内容
            * 支持代码高亮、目录等功能
            */}
          <MarkdownRenderer content={post.content} />
        </div>

        {/* 作者信息 */}
        <footer className="mt-12 pt-8 border-t">
          <div className="flex items-center gap-4">
            {post.author.image && (
              <img
                src={post.author.image}
                alt={post.author.name || ''}
                className="w-12 h-12 rounded-full"
              />
            )}
            <div>
              <p className="font-medium">{post.author.name}</p>
              <p className="text-sm text-muted-foreground">作者</p>
            </div>
          </div>
        </footer>
      </div>
    </article>
  )
}

/**
 * 关键学习点:
 *
 * 1. generateStaticParams:
```

#### 5.3 创建 Markdown 渲染组件

创建 `src/components/post/markdown-renderer.tsx`:

```typescript
/**
 * Markdown 渲染组件
 *
 * @技术栈:
 * - react-markdown: React Markdown 渲染器
 * - remark-gfm: GitHub Flavored Markdown 支持
 * - rehype-highlight: 代码高亮
 * - rehype-raw: 支持 HTML
 *
 * @功能:
 * 1. Markdown 渲染
 * 2. 代码语法高亮
 * 3. 表格、任务列表等 GFM 特性
 * 4. 自定义组件样式
 */

'use client'

import ReactMarkdown from 'react-markdown'
import remarkGfm from 'remark-gfm'
import rehypeHighlight from 'rehype-highlight'
import rehypeRaw from 'rehype-raw'
import 'highlight.js/styles/github-dark.css' // 代码高亮主题

interface MarkdownRendererProps {
  content: string
}

/**
 * MarkdownRenderer 组件
 *
 * @说明:
 * 将 Markdown 文本转换为 React 组件
 */
export function MarkdownRenderer({ content }: MarkdownRendererProps) {
  return (
    <ReactMarkdown
      /**
       * remarkPlugins: remark 插件
       *
       * @说明:
       * remark 处理 Markdown AST
       * 用于扩展 Markdown 语法
       *
       * @remarkGfm:
       * GitHub Flavored Markdown
       * 支持表格、任务列表、删除线等
       */
      remarkPlugins={[remarkGfm]}
      /**
       * rehypePlugins: rehype 插件
       *
       * @说明:
       * rehype 处理 HTML AST
       * 用于处理 HTML 输出
       *
       * @rehypeHighlight:
       * 代码语法高亮
       * 使用 highlight.js
       *
       * @rehypeRaw:
       * 允许在 Markdown 中使用原始 HTML
       */
      rehypePlugins={[rehypeHighlight, rehypeRaw]}
      /**
       * components: 自定义组件映射
       *
       * @说明:
       * 可以自定义 Markdown 元素的渲染方式
       * 替换默认的 HTML 标签
       */
      components={{
        /**
         * 代码块
         *
         * @说明:
         * 自定义代码块样式
         */
        code({ node, inline, className, children, ...props }) {
          /**
           * inline: 是否为行内代码
           *
           * @示例:
           * `code` -> inline = true
           * ```js\ncode\n``` -> inline = false
           */
          if (inline) {
            // 行内代码
            return (
              <code
                className="px-1.5 py-0.5 rounded bg-muted font-mono text-sm"
                {...props}
              >
                {children}
              </code>
            )
          }

          // 代码块
          return (
            <code className={className} {...props}>
              {children}
            </code>
          )
        },

        /**
         * 链接
         *

         */
        a({ href, children, ...props }) {

          return (
            <a
              href={href}
              className="text-primary hover:underline"
              /**
               * 外部链接安全属性
               *
               * @target="_blank":
               * 在新标签页打开
               *
               * @rel="noopener noreferrer":

               * - noopener: 防止新页面访问 window.opener

                target: '_blank',
                rel: 'noopener noreferrer',
              })}
              {...props}

      {content}
    </ReactMarkdown>
  )
}
```

```typescript
import { prisma } from '@/lib/prisma'

/**
 * 获取统计数据
 *
 * @说明:
 * 并行查询多个统计指标
 * 使用 Promise.all 提升性能
 */
async function getStats() {
  /**
   * 并行查询
   *
   * @技术原理:
   * Promise.all 同时执行多个异步操作
   * 等待所有操作完成后返回结果数组
   *
   * @性能:
   * 比串行查询快得多
   * 例如: 4个查询从 4秒 减少到 1秒
   */
  const [totalPosts, publishedPosts, draftPosts, totalViews, totalLikes, totalCategories, totalTags] =
    await Promise.all([
      // 文章总数
      prisma.post.count(),

      // 已发布文章数
      prisma.post.count({
        where: { status: 'published' },
      }),

      // 草稿文章数
      prisma.post.count({
        where: { status: 'draft' },
      }),

      /**
       * 总浏览量
       *
       * @技术原理:
       * aggregate 聚合查询
       * _sum 求和操作
       *
       * @SQL 等效:
       * SELECT SUM(views) FROM posts
       */
      prisma.post.aggregate({
        _sum: {
          views: true,
        },
      }),

      // 总点赞数
      prisma.post.aggregate({
        _sum: {
          likes: true,
        },
      }),

      // 分类数
      prisma.category.count(),

      // 标签数
      prisma.tag.count(),
    ])

  return {
    totalPosts,
    publishedPosts,
    draftPosts,
    totalViews: totalViews._sum.views || 0,
    totalLikes: totalLikes._sum.likes || 0,
    totalCategories,
    totalTags,
  }
}
```
