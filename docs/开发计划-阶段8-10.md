```typescript
/**
 * 主题提供者组件
 *
 * @技术原理:
 * 1. 使用 next-themes 库管理主题
 * 2. 支持亮色/暗色/系统自动
 * 3. 避免闪烁 (FOUC - Flash of Unstyled Content)
 *
 * @为什么使用 next-themes?
 * 1. SSR 友好: 避免闪烁
 * 2. 自动持久化: localStorage
 * 3. 系统主题检测
 * 4. TypeScript 支持
 */

'use client'

import * as React from 'react'
import { ThemeProvider as NextThemesProvider } from 'next-themes'
import { type ThemeProviderProps } from 'next-themes/dist/types'

/**
 * ThemeProvider 组件
 *
 * @说明:
 * 包装 next-themes 的 ThemeProvider
 * 添加项目特定的配置
 *
 * @参数:
 * - attribute: HTML 属性,用于应用主题 ('class' 或 'data-theme')
 * - defaultTheme: 默认主题
 * - enableSystem: 是否启用系统主题检测
 * - disableTransitionOnChange: 切换主题时禁用过渡动画
 *
 * @技术细节:
 * attribute="class" 会在 <html> 标签上添加 class="dark"
 * 配合 Tailwind CSS 的 dark: 修饰符使用
 */
export function ThemeProvider({ children, ...props }: ThemeProviderProps) {
  return (
    <NextThemesProvider
      attribute="class"
      defaultTheme="system"
      enableSystem
      disableTransitionOnChange
      {...props}
    >
      {children}
    </NextThemesProvider>
  )
}

/**
 * 避免闪烁的原理:
 *
 * 1. next-themes 会在 <head> 中注入一个 script
 * 2. 这个 script 在页面渲染前执行
 * 3. 读取 localStorage 中的主题设置
 * 4. 立即应用到 <html> 标签
 * 5. React 水合时,主题已经正确应用
 *
 * @对比传统方法:
 * 传统方法: 页面加载 -> React 水合 -> 读取主题 -> 应用主题 (闪烁)
 * next-themes: 页面加载 -> 应用主题 -> React 水合 (无闪烁)
 */
```

#### 8.2 实现搜索功能

创建 `src/app/(frontend)/search/page.tsx`:

```typescript
/**
 * 搜索页面
 *
 * @渲染策略: CSR (Client-Side Rendering)
 *
 * @原因:
 * 1. 搜索是交互式功能
 * 2. 需要实时更新结果
 * 3. 不需要 SEO (搜索结果页)
 *
 * @技术方案:
 * 1. 客户端发起搜索请求
 * 2. 调用 API Route
 * 3. 实时显示结果
 */

'use client'

import { useState, useEffect } from 'react'
import { useSearchParams } from 'next/navigation'
import { Search } from 'lucide-react'
import { Input } from '@/components/ui/input'
import { PostCard } from '@/components/post/post-card'

/**
 * 搜索页面组件
 */
export default function SearchPage() {
  /**
   * useSearchParams: 获取 URL 搜索参数
   *
   * @技术原理:
   * Next.js App Router 的 Hook
   * 读取 URL 的查询参数
   *
   * @示例:
   * /search?q=react -> searchParams.get('q') === 'react'
   */
  const searchParams = useSearchParams()
  const initialQuery = searchParams.get('q') || ''

  /**
   * 组件状态
   *
   * @useState:
   * React Hook,用于在函数组件中添加状态
   *
   * @状态:
   * - query: 搜索关键词
   * - results: 搜索结果
   * - loading: 加载状态
   */
  const [query, setQuery] = useState(initialQuery)
  const [results, setResults] = useState<any[]>([])
  const [loading, setLoading] = useState(false)

  /**
   * 搜索函数
   *
   * @说明:
   * 调用搜索 API,获取结果
   *
   * @技术原理:
   * 1. 使用 fetch API 发起请求
   * 2. 处理加载状态
   * 3. 更新结果
   */
  const handleSearch = async (searchQuery: string) => {
    if (!searchQuery.trim()) {
      setResults([])
      return
    }

    setLoading(true)

    try {
      /**
       * fetch: 浏览器原生 API
       *
       * @说明:
       * 发起 HTTP 请求
       *
       * @返回:
       * Promise<Response>
       */
      const response = await fetch(`/api/search?q=${encodeURIComponent(searchQuery)}`)

      if (!response.ok) {
        throw new Error('搜索失败')
      }

      const data = await response.json()
      setResults(data.results)
    } catch (error) {
      console.error('搜索错误:', error)
      // 这里可以显示错误提示
    } finally {
      setLoading(false)
    }
  }

  /**
   * useEffect: 副作用 Hook
   *
   * @技术原理:
   * 在组件渲染后执行副作用
   *
   * @依赖数组:
   * [initialQuery] - 当 initialQuery 变化时执行
   *
   * @用途:
   * 页面加载时,如果 URL 有搜索参数,自动执行搜索
   */
  useEffect(() => {
    if (initialQuery) {
      handleSearch(initialQuery)
    }
  }, [initialQuery])

  /**
   * 防抖搜索
   *
   * @技术原理:
   * 使用 setTimeout 延迟执行
   * 用户停止输入一段时间后才搜索
   *
   * @优势:
   * 1. 减少 API 请求
   * 2. 提升性能
   * 3. 更好的用户体验
   */
  useEffect(() => {
    const timer = setTimeout(() => {
      if (query !== initialQuery) {
        handleSearch(query)
      }
    }, 500) // 500ms 延迟

    /**
     * 清理函数
     *
     * @说明:
     * 组件卸载或下次执行前调用
     * 清除上一个 timer
     */
    return () => clearTimeout(timer)
  }, [query])

  return (
    <div className="container py-12">
      <div className="max-w-3xl mx-auto space-y-8">
        {/* 搜索标题 */}
        <div>
          <h1 className="text-4xl font-bold mb-4">搜索文章</h1>
          <p className="text-muted-foreground">
            输入关键词搜索相关文章
          </p>
        </div>

        {/* 搜索框 */}
        <div className="relative">
          <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 h-5 w-5 text-muted-foreground" />
          <Input
            type="search"
            placeholder="搜索..."
            value={query}
            /**
             * onChange: 输入变化事件
             *
             * @技术原理:
             * React 的受控组件
             * 输入值由 state 控制
             *
             * @流程:
             * 用户输入 -> 触发 onChange -> 更新 state -> 重新渲染
             */
            onChange={(e) => setQuery(e.target.value)}
            className="pl-10 text-lg"
            autoFocus
          />
        </div>

        {/* 搜索结果 */}
        {loading ? (
          <div className="text-center py-12">
            <div className="inline-block animate-spin rounded-full h-8 w-8 border-b-2 border-primary"></div>
            <p className="mt-4 text-muted-foreground">搜索中...</p>
          </div>
        ) : results.length > 0 ? (
          <div>
            <p className="text-sm text-muted-foreground mb-6">
              找到 {results.length} 篇相关文章
            </p>
            <div className="space-y-6">
              {results.map((post) => (
                <PostCard key={post.id} post={post} />
              ))}
            </div>
          </div>
        ) : query ? (
          <div className="text-center py-12">
            <p className="text-muted-foreground">未找到相关文章</p>
          </div>
        ) : null}
      </div>
    </div>
  )
}

/**
 * 关键学习点:
 *
 * 1. 客户端渲染:
 *    - 'use client' 指令
 *    - 可以使用 React Hooks
 *    - 可以处理用户交互
 *
 * 2. useState Hook:
 *    - 添加组件状态
 *    - 状态更新触发重新渲染
 *    - 每个组件实例独立状态
 *
 * 3. useEffect Hook:
 *    - 处理副作用
 *    - 依赖数组控制执行时机
 *    - 清理函数防止内存泄漏
 *
 * 4. 防抖技巧:
 *    - setTimeout 延迟执行
 *    - clearTimeout 取消上次执行
 *    - 提升性能
 *
 * 5. 加载状态:
 *    - 提供视觉反馈
 *    - 改善用户体验
 *    - 防止重复操作
 */
```

创建搜索 API:

```typescript
/**
 * 搜索 API Route
 *
 * @路径: /api/search
 *
 * @技术原理:
 * Next.js App Router 的 Route Handler
 * 处理服务端 API 请求
 */

import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'

/**
 * GET 请求处理函数
 *
 * @参数:
 * - request: NextRequest 对象
 *
 * @返回:
 * NextResponse 对象
 */
export async function GET(request: NextRequest) {
  /**
   * 获取查询参数
   *
   * @技术原理:
   * URLSearchParams API
   * 解析 URL 的查询字符串
   */
  const searchParams = request.nextUrl.searchParams
  const query = searchParams.get('q')

  // 验证查询参数
  if (!query || query.trim().length === 0) {
    return NextResponse.json(
      { error: '请提供搜索关键词' },
      { status: 400 }
    )
  }

  try {
    /**
     * 全文搜索
     *
     * @技术原理:
     * Prisma 的 OR 条件和 contains 过滤
     *
     * @局限性:
     * 这是简单的模糊匹配,不是真正的全文搜索
     * 对于大型应用,建议使用:
     * - Algolia
     * - Elasticsearch
     * - PostgreSQL 全文搜索
     *
     * @查询字段:
     * - title: 标题
     * - content: 内容
     * - excerpt: 摘要
     */
    const results = await prisma.post.findMany({
      where: {
        status: 'published',
        OR: [
          { title: { contains: query, mode: 'insensitive' } },
          { content: { contains: query, mode: 'insensitive' } },
          { excerpt: { contains: query, mode: 'insensitive' } },
        ],
      },
      take: 20, // 限制结果数量
      orderBy: { publishedAt: 'desc' },
      include: {
        author: {
          select: {
            name: true,
            image: true,
          },
        },
        category: true,
        tags: true,
      },
    })

    /**
     * 返回 JSON 响应
     *
     * @技术原理:
     * NextResponse.json() 自动:
     * 1. 序列化数据
     * 2. 设置 Content-Type: application/json
     * 3. 返回正确的 HTTP 响应
     */
    return NextResponse.json({
      results,
      count: results.length,
    })
  } catch (error) {
    console.error('搜索错误:', error)
    return NextResponse.json(
      { error: '搜索失败,请稍后重试' },
      { status: 500 }
    )
  }
}

/**
 * 搜索优化建议:
 *
 * 1. 使用专业搜索服务:
 *    - Algolia: 快速,易用,有免费额度
 *    - Meilisearch: 开源,可自托管
 *    - Elasticsearch: 功能强大,适合大型应用
 *
 * 2. 数据库全文搜索:
 *    - PostgreSQL: 内置全文搜索
 *    - MySQL: FULLTEXT 索引
 *    - MongoDB: Text 索引
 *
 * 3. 搜索高亮:
 *    - 在结果中高亮关键词
 *    - 提供上下文片段
 *
 * 4. 搜索建议:
 *    - 自动补全
 *    - 拼写纠正
 *    - 热门搜索
 */
```

```typescript
/**
 * 生成网站地图
 *
 * @技术原理:
 * Next.js 13+ 支持动态生成 sitemap.xml
 * 搜索引擎会定期抓取这个文件
 *
 * @SEO 作用:
 * 1. 帮助搜索引擎发现所有页面
 * 2. 提供页面优先级和更新频率
 * 3. 加快索引速度
 *
 * @访问路径:
 * https://yourdomain.com/sitemap.xml
 */

import { MetadataRoute } from 'next'
import { prisma } from '@/lib/prisma'
import { siteConfig } from '@/config/site'

/**
 * sitemap 函数
 *
 * @返回:
 * Sitemap 对象数组
 *
 * @执行时机:
 * 1. 构建时生成 (SSG)
 * 2. 可以配置 ISR 定期重新生成
 */
export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  // 获取所有已发布的文章
  const posts = await prisma.post.findMany({
    where: { status: 'published' },
    select: {
      slug: true,
      updatedAt: true,
    },
    orderBy: { updatedAt: 'desc' },
  })

  // 获取所有分类
  const categories = await prisma.category.findMany({
    select: {
      slug: true,
      updatedAt: true,
    },
  })

  // 获取所有标签
  const tags = await prisma.tag.findMany({
    select: {
      slug: true,
      updatedAt: true,
    },
  })

  /**
   * 静态页面
   *
   * @优先级:
   * 1.0: 最高 (首页)
   * 0.8: 高 (主要页面)
   * 0.6: 中 (次要页面)
   */
  const staticPages = [
    {
      url: siteConfig.url,
      lastModified: new Date(),
      changeFrequency: 'daily' as const,
      priority: 1,
    },
    {
      url: `${siteConfig.url}/posts`,
      lastModified: new Date(),
      changeFrequency: 'daily' as const,
      priority: 0.8,
    },
    {
      url: `${siteConfig.url}/about`,
      lastModified: new Date(),
      changeFrequency: 'monthly' as const,
      priority: 0.5,
    },
  ]

  /**
   * 文章页面
   *
   * @优先级: 0.7
   * @更新频率: weekly
   */
  const postPages = posts.map((post) => ({
    url: `${siteConfig.url}/posts/${post.slug}`,
    lastModified: post.updatedAt,
    changeFrequency: 'weekly' as const,
    priority: 0.7,
  }))

  // 分类页面
  const categoryPages = categories.map((category) => ({
    url: `${siteConfig.url}/categories/${category.slug}`,
    lastModified: category.updatedAt,
    changeFrequency: 'weekly' as const,
    priority: 0.6,
  }))

  // 标签页面
  const tagPages = tags.map((tag) => ({
    url: `${siteConfig.url}/tags/${tag.slug}`,
    lastModified: tag.updatedAt,
    changeFrequency: 'weekly' as const,
    priority: 0.5,
  }))

  /**
   * 合并所有页面
   *
   * @返回:
   * 完整的 sitemap 数据
   */
  return [...staticPages, ...postPages, ...categoryPages, ...tagPages]
}

/**
 * Sitemap 字段说明:
 *
 * 1. url: 页面的完整 URL
 * 2. lastModified: 最后修改时间
 * 3. changeFrequency: 更新频率
 *    - always: 每次访问都变化
 *    - hourly: 每小时
 *    - daily: 每天
 *    - weekly: 每周
 *    - monthly: 每月
 *    - yearly: 每年
 *    - never: 从不
 * 4. priority: 优先级 (0.0 - 1.0)
 *    - 1.0: 最重要
 *    - 0.5: 普通
 *    - 0.1: 不重要
 */
```

```typescript
async function getPost(slug: string) {

     * 永久缓存,除非手动重新验证

    cache: 'force-cache',

    cache: 'no-store',
```
